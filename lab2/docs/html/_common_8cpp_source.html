40long Date::toDayNumber() const {
41 // Алгоритм days_from_civil (H. Hinnant).
42 int y = year_ - (month_ <= 2 ? 1 : 0);
43 const long era = (y >= 0 ? y : y - 399) / 400;
44 const unsigned yoe = static_cast<unsigned>(y - era * 400);
45 const unsigned m = static_cast<unsigned>(month_);
46 const unsigned doy = (153 * (m > 2 ? m - 3 : m + 9) + 2) / 5 + static_cast<unsigned>(day_) - 1;
47 const unsigned doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
48 return era * 146097 + static_cast<long>(doe) - 719468;
49}

51Date Date::fromDayNumber(long z) {
52 z += 719468;
53 const long era = (z >= 0 ? z : z - 146096) / 146097;
54 const unsigned doe = static_cast<unsigned>(z - era * 146097);
55 const unsigned yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
56 const long y = static_cast<long>(yoe) + era * 400;
57 const unsigned doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
58 const unsigned mp = (5 * doy + 2) / 153;
59 const unsigned d = doy - (153 * mp + 2) / 5 + 1;
60 const unsigned m = mp < 10 ? mp + 3 : mp - 9;
61 return Date(static_cast<int>(d), static_cast<int>(m), static_cast<int>(y + (m <= 2 ? 1 : 0)));
62}
