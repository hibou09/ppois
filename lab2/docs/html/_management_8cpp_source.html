108void Budget::spend(const std::string& category, double amount) {
109 if (category.empty() || amount <= 0.0) {
110 throw InvalidParameterException("expense needs category and positive amount");
111 }
112 if (expenses_ + amount > expenseLimit_) {
113 throw BudgetExceededException(category + ": limit " + std::to_string(expenseLimit_));
114 }
115 if (amount > balance_) {
116 throw InsufficientFundsException(amount, balance_);
117 }
118 balance_ -= amount;
119 expenses_ += amount;
120 byCategory_[category] += amount;
121}
122
123double Budget::spentOn(const std::string& category) const {
124 auto it = byCategory_.find(category);
125 return it == byCategory_.end() ? 0.0 : it->second;
126}

153double Payroll::payAll(const std::vector<Employee*>& employees, Budget& budget) {
154 double total = 0.0;
155 for (const Employee* e : employees) {
156 if (e != nullptr && e->isActive()) {
157 total += calculate(*e);
158 }
159 }
160 if (total <= 0.0) {
161 return 0.0;
162 }
163 budget.spend("salary", total); // всё или ничего
164 for (const Employee* e : employees) {
165 if (e != nullptr && e->isActive()) {
166 ++paymentsCount_;
167 }
168 }
169 totalPaid_ += total;
170 return total;
171}

303int Oceanarium::runDailyOperations(const Date& today) {
304 int created = 0;
305 for (auto& t : tanks_) {
306 try {
307 t->runLifeSupport(24.0);
308 } catch (const EquipmentFailureException& e) {
309 maintenance_.schedule(std::string("Repair: ") + e.what(), t.get(), 5, today);
310 ++created;
311 }
312 for (const auto& a : t->animals()) {
313 a->passDay();
314 }
315 if (t->needsCleaning()) {
316 maintenance_.schedule("Clean " + t->name(), t.get(), 4, today.addDays(1));
317 ++created;
318 }
319 if (t->filtration().needsMaintenance()) {
320 maintenance_.schedule("Replace filter in " + t->name(), t.get(), 3, today.addDays(2));
321 ++created;
322 }
323 }
324 foodStorage_.removeExpired(today);
325 return created;
326}
