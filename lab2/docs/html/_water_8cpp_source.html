59void WaterParameters::mixWith(const WaterParameters& fresh, double fraction) {
60 if (fraction < 0.0 || fraction > 1.0) {
61 throw InvalidParameterException("fraction must be in [0, 1]");
62 }
63 auto mix = [fraction](double a, double b) { return a * (1.0 - fraction) + b * fraction; };
64 temperature_ = mix(temperature_, fresh.temperature_);
65 salinity_ = mix(salinity_, fresh.salinity_);
66 ph_ = mix(ph_, fresh.ph_);
67 oxygen_ = mix(oxygen_, fresh.oxygen_);
68 ammonia_ = mix(ammonia_, fresh.ammonia_);
69}

75double WaterParameters::qualityIndex() const {
76 double score = 100.0;
77 score -= ammonia_ * 100.0;
78 score -= std::fabs(ph_ - 8.1) * 10.0;
79 if (oxygen_ < 6.0) {
80 score -= (6.0 - oxygen_) * 10.0;
81 }
82 return std::clamp(score, 0.0, 100.0);
83}

131double WaterRequirements::deviation(const WaterParameters& w) const {
132 auto outside = [](double v, double lo, double hi) {
133 if (v < lo) return lo - v;
134 if (v > hi) return v - hi;
135 return 0.0;
136 };
137 double result = outside(w.temperature(), minTemp_, maxTemp_) +
138 outside(w.salinity(), minSalinity_, maxSalinity_) +
139 outside(w.ph(), minPh_, maxPh_);
140 if (w.oxygen() < minOxygen_) {
141 result += minOxygen_ - w.oxygen();
142 }
143 return result;
144}

160double Sensor::read(const WaterParameters& water) {
161 if (!calibrated_) {
162 throw SensorNotCalibratedException(id_);
163 }
164 switch (type_) {
165 case SensorType::Temperature: lastValue_ = water.temperature(); break;
166 case SensorType::Salinity: lastValue_ = water.salinity(); break;
167 case SensorType::Ph: lastValue_ = water.ph(); break;
168 case SensorType::Oxygen: lastValue_ = water.oxygen(); break;
169 case SensorType::Ammonia: lastValue_ = water.ammonia(); break;
170 }
171 if (++readingsSinceCalibration_ >= kDriftLimit) {
172 calibrated_ = false;
173 }
174 return lastValue_;
175}

195void FiltrationSystem::runCycle(WaterParameters& water, double hours) {
196 if (hours <= 0.0) {
197 throw InvalidParameterException("filtration hours must be positive");
198 }
199 if (!active_) {
200 throw EquipmentFailureException("filtration system " + model_ + " is off");
201 }
202 if (filterWear_ >= 100.0) {
203 throw EquipmentFailureException("filter of " + model_ + " is clogged");
204 }
205 double efficiency = 1.0 - filterWear_ / 100.0;
206 water.reduceAmmonia(std::min(0.9, 0.2 * hours * efficiency));
207 filterWear_ = std::min(100.0, filterWear_ + hours * 0.5);
208}
