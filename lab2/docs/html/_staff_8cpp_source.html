57void Aquarist::assignTank(Tank* tank) {
58 if (tank == nullptr) {
59 throw InvalidParameterException("tank must not be null");
60 }
61 if (isResponsibleFor(tank->id())) {
62 throw InvalidParameterException("tank already assigned to " + name_);
63 }
64 if (static_cast<int>(assignedTanks_.size()) >= certLevel_ * 2) {
65 throw EmployeeNotQualifiedException(name_ + " cannot maintain more than " +
66 std::to_string(certLevel_ * 2) + " tanks");
67 }
68 if (dynamic_cast<CoralReefTank*>(tank) != nullptr && certLevel_ < 2) {
69 throw EmployeeNotQualifiedException("coral reef tanks require level 2 certification");
70 }
71 assignedTanks_.push_back(tank);
72}

77}

341int StaffSchedule::removeShiftsOf(int employeeId) {

343 shifts_.erase(std::remove_if(shifts_.begin(), shifts_.end(),
