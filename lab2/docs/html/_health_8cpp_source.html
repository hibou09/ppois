81double Medication::dispense(const Animal& animal, const Date& today) {
82 if (isExpired(today)) {
83 throw ExpiredProductException(name_ + " expired on " + expiry_.toString());
84 }
85 double dose = doseFor(animal);
86 if (dose > stockMg_) {
87 throw OutOfStockException(name_ + ": need " + std::to_string(dose) + " mg");
88 }
89 stockMg_ -= dose;
90 return dose;
91}
