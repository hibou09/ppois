72int FeedingSchedule::execute(int minute, const std::vector<std::shared_ptr<Animal>>& animals,
73 FoodStorage& storage, const Date& today) {
74 int fed = 0;
75 for (const auto& entry : dueAt(minute)) {
76 for (const auto& animal : animals) {
77 if (animal && animal->id() == entry.animalId()) {
78 animal->feed(storage.find(entry.foodName()), entry.amountKg(), today);
79 ++fed;
80 ++executedCount_;
81 }
82 }
83 }

97double Supplier::quote(double kg, double pricePerKg) const {
98 if (kg <= 0.0 || pricePerKg < 0.0) {
99 throw InvalidParameterException("quote needs positive amount and non-negative price");

118void Supplier::rate(int score) {

120 throw InvalidParameterException("rating must be in [1, 5]");
