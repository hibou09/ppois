31void Tank::addAnimal(std::shared_ptr<Animal> animal) {
32 if (!animal) {
33 throw InvalidParameterException("animal must not be null");
34 }
35 if (hasAnimal(animal->id())) {
36 throw InvalidParameterException("animal #" + std::to_string(animal->id()) +
37 " already lives in " + name_);
38 }
39 if (animalCount() >= maxAnimals_) {
40 throw TankCapacityExceededException(name_ + " holds at most " +
41 std::to_string(maxAnimals_) + " animals");
42 }
43 if (!animal->canLiveIn(water_)) {
44 throw InvalidWaterParametersException("water in " + name_ + " is unsuitable for " +
45 animal->name());
46 }
47 if (!canHost(*animal)) {
48 throw IncompatibleSpeciesException(type() + " cannot host " + animal->describe());
49 }
50 for (const auto& resident : animals_) {
51 if (!animal->canShareTankWith(*resident)) {
52 throw IncompatibleSpeciesException(animal->name() + " cannot live with " +
53 resident->name());
54 }
55 }
56 animals_.push_back(std::move(animal));
57}

59std::shared_ptr<Animal> Tank::removeAnimal(int animalId) {

130void Tank::runLifeSupport(double hours) {
131 if (hours <= 0.0) {
132 throw InvalidParameterException("hours must be positive");
133 }
134 water_.addAmmonia(totalBiomass() * hours * 0.002 / (volumeLiters_ / 1000.0));
135 heating_.regulate(water_, hours);
136 filtration_.runCycle(water_, hours);
