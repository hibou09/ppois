171 : basePrice_(basePrice), dailyLimit_(dailyLimit), revenue_(0.0), nextTicketId_(1),
172 refundsCount_(0) {
173 if (basePrice <= 0.0 || dailyLimit <= 0) {

178double TicketOffice::priceFor(const Visitor& visitor, TicketType type) const {
179 switch (type) {
180 case TicketType::Adult:
181 return basePrice_;
182 case TicketType::Child:
183 if (!visitor.isChild()) {
184 throw InvalidTicketException(visitor.name() + " is not eligible for child ticket");
185 }
186 return basePrice_ * 0.5;
187 case TicketType::Senior:
188 if (!visitor.isSenior()) {
189 throw InvalidTicketException(visitor.name() +
190 " is not eligible for senior ticket");
191 }
192 return basePrice_ * 0.6;
193 case TicketType::Family:
194 return basePrice_ * 3.0;
195 }
196 return basePrice_;
197}
198
199Ticket TicketOffice::sell(Visitor& visitor, TicketType type, const Date& date,
200 const Membership* membership, PromoCode* promo) {
201 int persons = type == TicketType::Family ? 4 : 1;
202 if (soldOn(date) + persons > dailyLimit_) {
203 throw TicketSoldOutException("no tickets left for " + date.toString());
204 }
205 double price = priceFor(visitor, type);
206 if (membership != nullptr && membership->visitorId() == visitor.id()) {
207 price = membership->applyDiscount(price, date);
208 }
209 if (promo != nullptr) {
210 if (!promo->isValid(date)) {
211 throw InvalidPromoCodeException(promo->code());
212 }
213 price = promo->discounted(price);
214 }
215 visitor.pay(price); // может бросить InsufficientFundsException
216 if (promo != nullptr) {
217 promo->apply(price, date);
218 }
219 soldPerDay_[date.toDayNumber()] += persons;
220 revenue_ += price;
221 return Ticket(nextTicketId_++, type, price, date, visitor.id());
222}
223
224void TicketOffice::refund(Ticket& ticket, Visitor& visitor, const Date& today) {
225 if (ticket.visitorId() != visitor.id()) {
226 throw InvalidTicketException("ticket belongs to another visitor");
227 }
228 if (!ticket.isRefundable(today)) {
229 throw InvalidTicketException("ticket #" + std::to_string(ticket.id()) +
230 " cannot be refunded");
231 }
232 if (ticket.price() > 0.0) {
233 visitor.topUp(ticket.price());
234 }
235 revenue_ -= ticket.price();
236 soldPerDay_[ticket.visitDate().toDayNumber()] -= ticket.personCount();
237 ticket.invalidate();
238 ++refundsCount_;
239}
240
241int TicketOffice::soldOn(const Date& date) const {
242 auto it = soldPerDay_.find(date.toDayNumber());
243 return it == soldPerDay_.end() ? 0 : it->second;
244}
245
246int TicketOffice::remainingFor(const Date& date) const {
247 return dailyLimit_ - soldOn(date);
248}
