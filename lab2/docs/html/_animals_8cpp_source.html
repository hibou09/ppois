349double Coral::grow(const WaterParameters& water) {
350 if (water.temperature() > kBleachingTemperature) {
351 healthPercent_ = std::max(0.0, healthPercent_ - 10.0);
352 if (healthPercent_ < 50.0) {
353 bleached_ = true;

364 }
365 sizeCm2_ -= cm2;

367}
